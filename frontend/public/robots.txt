# AI crawler policy for GeoStrategy. This product helps a site appear in AI
# search and AI answers, so this file names and welcomes the crawlers that
# do that. It does not yet state a preference for the crawlers that train
# AI models — see the commented-out block near the end of this file.

User-agent: *
# Content-Signal states our preference for how a crawler may use content it
# is already allowed to fetch, per Cloudflare's Content Signals Policy.
# search=yes allows search indexing. use=reference allows an excerpt with a
# link back. ai-input=yes allows a model to read a page to answer a live
# question, which is exactly how this site wants to be found. ai-train=no
# declines use as AI model training data, which is a separate thing.
# Exact syntax confirmed from Cloudflare's own docs:
# https://developers.cloudflare.com/bots/additional-configurations/managed-robots-txt/
Content-signal: search=yes, ai-input=yes, use=reference, ai-train=no
Allow: /
Allow: /r/
Disallow: /dashboard
Disallow: /assessments/
Disallow: /sites/
Disallow: /account
Disallow: /auth/
Disallow: /verify-email
Disallow: /reset-password
